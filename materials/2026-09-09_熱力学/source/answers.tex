\pagehead{第1問\quad 三角形のサイクル\quad 〔温度・仕事〕}
\begin{minipage}[t]{.59\textwidth}\vspace{0pt}\raggedright
\textbf{(1)} $pV=nRT$ より、$T\propto pV$ である。
\[
\boxed{T_{\mathrm B}=2T_0},\qquad \boxed{T_{\mathrm C}=3T_0}.
\]
\textbf{(2)} AB は定積変化なので仕事は0。BC の仕事は台形の面積、CA は圧縮なので負である。
\begin{align*}
W_{\mathrm{AB}}&=\boxed{0},\\
W_{\mathrm{BC}}&=\frac{2p_0+p_0}{2}(3V_0-V_0)=\boxed{3p_0V_0},\\
W_{\mathrm{CA}}&=p_0(V_0-3V_0)=\boxed{-2p_0V_0}.
\end{align*}
\end{minipage}\hfill

\pagehead{第1問\quad 三角形のサイクル\quad 〔熱量・効率〕}
\textbf{(2)} $\displaystyle\Delta U=\frac32nR\Delta T=\frac32\Delta(pV)$、$Q=\Delta U+W$ より、
\begin{center}
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{cccc}
\toprule 過程 & $W$ & $\Delta U$ & $Q$\\\midrule
A$\to$B & $0$ & $\frac32p_0V_0$ & $\frac32p_0V_0$\\
B$\to$C & $3p_0V_0$ & $\frac32p_0V_0$ & $\frac92p_0V_0$\\
C$\to$A & $-2p_0V_0$ & $-3p_0V_0$ & $-5p_0V_0$\\\bottomrule
\end{tabular}
\end{center}
\textbf{(3)} 1周の仕事は三角形の面積に等しい。AB と BC で吸熱し、CA で放熱する。
\[
\boxed{W_{\mathrm{net}}=\frac12(p_0)(2V_0)=p_0V_0},\qquad
Q_{\mathrm{in}}=\frac32p_0V_0+\frac92p_0V_0=6p_0V_0.
\]
\[
\boxed{\eta=\frac{W_{\mathrm{net}}}{Q_{\mathrm{in}}}=\frac16\simeq16.7\%}.
\]

\pagehead{第2問\quad 鉛直ピストン\quad 〔つり合い・各状態〕}
\textbf{(1)} ピストンに働く上向きの力は $p_0S$、下向きの力は $p_{\mathrm a}S+Mg$。
\[
p_0S=p_{\mathrm a}S+Mg
\quad\Longrightarrow\quad
\boxed{p_0=p_{\mathrm a}+\frac{Mg}{S}}.
\]
\textbf{(2)} 操作 I では、大気圧とピストンの重さが一定なので圧力も $p_0$ のままである。\\
体積が2倍になるため、状態方程式から温度も2倍になる。
\[
\boxed{p_{\mathrm B}=p_0,\qquad T_{\mathrm B}=2T_0}.
\]
操作 II は体積が $2V_0$ で一定であり、温度が $2T_0$ から $4T_0$ になる。
\[
\boxed{p_{\mathrm C}=2p_0,\qquad T_{\mathrm C}=4T_0}.
\]
\vfill
\note{固定を外す直前には $p_{\mathrm C}S=p_{\mathrm a}S+Mg+F$ とするため、$F=p_0S$ に調整する。}

\pagehead{第2問\quad 鉛直ピストン\quad 〔熱量・断熱圧縮〕}
{\fontsize{18}{24}\selectfont
\textbf{(3) 操作 I（定圧膨張）}
\[
W_{\mathrm I}=p_0(2V_0-V_0)=\boxed{p_0V_0},\qquad
Q_{\mathrm I}=\frac32nR(T_0)+p_0V_0=\boxed{\frac52p_0V_0}.
\]
\textbf{操作 II（定積加熱）}
\[
\boxed{W_{\mathrm{II}}=0},\qquad
Q_{\mathrm{II}}=\frac32nR(4T_0-2T_0)=\boxed{3p_0V_0}.
\]
\textbf{(4)} $pV^{5/3}=\text{一定}$ と $pV=nRT$ から、$TV^{2/3}=\text{一定}$。
\[
4T_0(2V_0)^{2/3}=T_{\mathrm D}V_0^{2/3}
\quad\Longrightarrow\quad\boxed{T_{\mathrm D}=4\cdot2^{2/3}T_0}.
\]
断熱変化では $Q=0$ なので、気体がされた仕事は内部エネルギーの増加に等しい。
\[
W_{\mathrm{on}}=\Delta U
=\frac32nR(T_{\mathrm D}-4T_0)
=\boxed{6p_0V_0\left(2^{2/3}-1\right)}.
\]
\par}
\vfill
\note{ここで $W_{\mathrm{on}}$ は外部から気体にされた仕事の総量。追加の外力 $F$ だけの仕事ではない。}

\pagehead{第3問\quad 状態方程式と内部エネルギー\quad 〔解答〕}
\textbf{(1)} 初めの温度と内部エネルギーは、
\[
T=\frac{pV}{nR}
=\frac{(1.0\times10^5)(1.2\times10^{-2})}{0.50\times8.0}
=\boxed{3.0\times10^2\,\mathrm K},
\]
\[
U=\frac32nRT=\frac32pV
=\frac32\times1.2\times10^3
=\boxed{1.8\times10^3\,\mathrm J}.
\]
\textbf{(2)} 体積一定では $T\propto p$ である。
\[
T'=T\frac{p'}p=300\times\frac{1.5\times10^5}{1.0\times10^5}
=\boxed{4.5\times10^2\,\mathrm K},
\]
\[
\Delta U=\frac32nR(T'-T)
=\frac32\times0.50\times8.0\times150
=\boxed{9.0\times10^2\,\mathrm J}.
\]

\pagehead{第4問\quad 熱気球が浮く条件\quad 〔解答〕}
{\fontsize{18}{24}\selectfont
\textbf{(1)} 内外の圧力は等しく、同じ空気なので $\rho T=\rho_0T_0$。
\[
\rho=\rho_0\frac{T_0}{T}=1.2\times\frac{300}{400}
=\boxed{0.90\,\mathrm{kg/m^3}}.
\]
\textbf{(2)} 浮力は外気の密度を使って求める。
\[
\boxed{F_{\mathrm b}=\rho_0Vg=1.2\times10^3\times10=1.2\times10^4\,\mathrm N}.
\]
全重量は $(M+\rho V)g=(300+900)\times10=1.2\times10^4\,\mathrm N$ だから、
\[
\boxed{F_{\mathrm{net}}=F_{\mathrm b}-(M+\rho V)g=0\,\mathrm N}.
\]
\textbf{(3)} 静止状態から浮き上がる条件は、浮力が全重量より大きいこと。
\[
\rho_0Vg>\left(M+\rho_0\frac{T_0}{T}V\right)g
\quad\Longrightarrow\quad
T>\frac{300}{1-\frac{300}{1.2\times1000}}
=\boxed{400\,\mathrm K}.
\]
\par}
\vfill
\note{$T=400\,\mathrm K$ ではつり合うだけなので、浮上には $T>400\,\mathrm K$ が必要。\\
下部が開いているため、加熱中に内部の空気の質量は一定ではない。}
